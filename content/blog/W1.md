# 1. 实现方法

DrawShader：鼠标点击时，在当前高度图上绘制一个区域作为波源能量输入。

RippleShader：每帧将高度图作为波形数据进行迭代计算，模拟水波扩散。

# 2. 离散波动方程
RippleShader中的计算 d = (上下左右之和)/2 - 上一帧中心值，本质是二维波动方程的显式中心差分求解。它利用当前帧的邻居高度与上一帧的历史高度，推算出下一帧的波形状态，乘上 0.99 阻尼系数模拟能量耗散。

# 3. 运行流程
每帧执行严格的三步逻辑：
绘制（点击触发，叠加凸起到当前帧）
计算（利用当前帧和上一帧，渲染出下一帧到临时纹理）
交换（更新指针：旧帧变为上一帧，新帧变为当前帧并显示）

# 4. 总结
全程在 GPU 着色器中完成，无需粒子系统，性能极高。利用 RFloat 格式纹理存储浮点高度值，并通过交换 RenderTexture 引用避免昂贵的内存拷贝。

# 5.代码实现

```c#
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Ripple : MonoBehaviour
{
    public Camera mainCamera;//获取摄像机
    public RenderTexture PrevRT;//上帧
    public RenderTexture CurrentRT;//当前帧
    public RenderTexture TempRT;//临时帧
    
    public Shader DrawShader;//绘制shader
    public Shader RippleShader;//涟漪计算shader
    private Material RippleMat;//生成涟漪的高度图
    private Material DrawMat;
    public int TextureSize = 512;

    [Range(0,1.0f)]
    public float DrawRadius = 0.2f;
    // Start is called before the first frame update
    void Start()
    {
        
        mainCamera = Camera.main.GetComponent<Camera>();
        CurrentRT = CreateRT();
        TempRT = CreateRT();
        PrevRT = CreateRT();

        DrawMat = new Material(DrawShader);
        RippleMat = new Material(RippleShader);
        GetComponent<Renderer>().material.mainTexture = CurrentRT;
    }

    public RenderTexture CreateRT()
    {
        RenderTexture rt = new RenderTexture(TextureSize,TextureSize,0,RenderTextureFormat.RFloat);
        rt.Create();
        return rt;
    }   




    /// <summary>
    /// 绘制
    /// </summary>
    /// <param name="x"></param>
    /// <param name="y"></param>
    /// <param name="radius"></param>
    private void DrawAt(float x,float y,float radius)
    {
        //原贴图，记住之前的累积绘制图CurrentRT
        DrawMat.SetTexture("_SourceTex",CurrentRT);
        //绘制位置&大小，传给材质这次点击的位置
        DrawMat.SetVector("_Pos",new Vector4(x,y,radius));
        //提交，用材质渲染一张新图到TempRT
        Graphics.Blit(null,TempRT,DrawMat);

        //进行交换，把原贴图CurrentRT替换成新的TempRT
        RenderTexture rt = TempRT;
        TempRT = CurrentRT;
        CurrentRT = rt;
    }







    // Update is called once per frame
    void Update()
    {
        if (Input.GetMouseButton(0))
        {
            //鼠标按下时射线检测
            Ray ray = mainCamera.ScreenPointToRay(Input.mousePosition);
            RaycastHit hit;//射线碰撞信息
            if(Physics.Raycast(ray, out hit))
            { 
                //TODO:绘制图案
                DrawAt(hit.textureCoord.x,hit.textureCoord.y,DrawRadius);
            }
        }
        //TODO：涟漪计算
        RippleMat.SetTexture("_PrevRT",PrevRT);
        RippleMat.SetTexture("_CurrentRT",CurrentRT);
        Graphics.Blit(null,TempRT,RippleMat);
        //TODO:交换到currentrt上，保存currentrt为最新帧
        Graphics.Blit(TempRT,PrevRT);
        RenderTexture rt = PrevRT;
        PrevRT = CurrentRT;
        CurrentRT = rt;

    }



    
       
}
```


```shader
Shader "Unlit/DrawShader"
{
    Properties
    {

    }




    SubShader
    {
        Tags { "RenderType"="Opaque" }
        LOD 100

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            
            #include "UnityCG.cginc"

            struct appdata
            {
                float4 vertex : POSITION;
                float2 uv : TEXCOORD0;
            };

            sampler2D _SourceTex;
            float4 _Pos;

            struct v2f
            {
                float2 uv : TEXCOORD0;
                float4 vertex : SV_POSITION;
            };


            v2f vert (appdata v)
            {
                v2f o;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = v.uv;

                return o;
            }

            fixed4 frag (v2f i) : SV_Target
            {
                float2 uv = i.uv;

                //return length(uv - _Pos.xy)/_Pos.z;//pos.z是drawmat函数的drawRadius
                return max(_Pos.z - length(uv - _Pos.xy)/_Pos.z,0) + tex2D(_SourceTex,uv).x;//最后新值 + 旧值加起来，实现每次点击叠加到上一次的结果上，形成累积效果
            }
            ENDCG
        }
    }
}
```

```shader
Shader "Unlit/RippleShader"
{
    Properties
    {

    }
    SubShader
    {
        Tags { "RenderType"="Opaque" }
        LOD 100

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag

            #include "UnityCG.cginc"

            sampler2D _PrevRT;//上帧
            sampler2D _CurrentRT;//当前帧
            float4 _CurrentRT_TexelSize;//贴图大小信息


            struct appdata
            {
                float4 vertex : POSITION;
                float2 uv : TEXCOORD0;
            };

            struct v2f
            {
                float2 uv : TEXCOORD0;

                float4 vertex : SV_POSITION;
            };



            v2f vert (appdata v)
            {
                v2f o;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = v.uv;
                return o;
            }

            fixed4 frag (v2f i) : SV_Target
            {
                //最小偏移单位
                float3 e = float3(_CurrentRT_TexelSize.xy,0);
                float2 uv = i.uv;

                //获取当前帧的上下左右的值
                float p10 = tex2D(_CurrentRT,uv - e.zy).x;//下
                float p01 = tex2D(_CurrentRT,uv - e.xz).x;//左
                float p21 = tex2D(_CurrentRT,uv + e.xz).x;//右
                float p12 = tex2D(_CurrentRT,uv + e.zy).x;//上
                //上一帧的中心值
                float p11 = tex2D(_PrevRT,uv).x;

                float d = (p10 + p01 + p21 + p12)/2 - p11;
                //加上细微衰减
                d *= 0.99;
                return d;
            }
            ENDCG
        }
    }
}
```
